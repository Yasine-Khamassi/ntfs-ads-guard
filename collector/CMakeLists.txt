cmake_minimum_required(VERSION 3.20)
project(ads_guard_collector LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "ADS Guard collector requires Windows.")
endif()

add_library(ads-guard-core STATIC analysis.cpp config.cpp)
target_compile_features(ads-guard-core PUBLIC cxx_std_17)
target_include_directories(ads-guard-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ../driver)
target_compile_definitions(ads-guard-core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ads-guard-core PUBLIC bcrypt)
if(MSVC)
  target_compile_options(ads-guard-core PUBLIC /W4 /WX /permissive- /utf-8)
endif()

add_executable(ads-guard-collector main.cpp)
target_link_libraries(ads-guard-collector PRIVATE ads-guard-core FltLib)

enable_testing()
add_executable(ads-guard-unit-tests tests/unit_tests.cpp)
target_link_libraries(ads-guard-unit-tests PRIVATE ads-guard-core)
add_test(NAME unit COMMAND ads-guard-unit-tests)
