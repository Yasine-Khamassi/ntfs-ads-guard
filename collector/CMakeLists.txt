cmake_minimum_required(VERSION 3.20)
project(ads_guard_collector LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "ADS Guard collector requires Windows.")
endif()
add_executable(ads-guard-collector main.cpp config.cpp)
target_compile_features(ads-guard-collector PRIVATE cxx_std_17)
target_include_directories(ads-guard-collector PRIVATE ../driver)
target_compile_definitions(ads-guard-collector PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ads-guard-collector PRIVATE FltLib)
